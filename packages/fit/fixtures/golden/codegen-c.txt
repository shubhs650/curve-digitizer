==== spline ====
/*
 * Curve Digitizer fit: Cubic spline, 7 knots
 * Dataset: Torque
 * Valid for 0 <= x <= 10, the range of the 80 fitted points; outside it the curve is extrapolated
 * R^2 = 0.99841, RMSE = 0.0287, AICc = -540.3
 */
#include <math.h>

double torque(double x)
{
    static const double knots[] = {0.0, 1.25, 2.5, 4.0, 5.5, 8.0, 10.0};
    static const double values[] = {1.0, 2.1, 1.6, 0.2, -1.05, 0.5, 0.92};
    static const double slopes[] = {
        1.189899854508072, 0.26020029098385655, -0.7907010184434978, -1.073155868029238,
        -0.21667550943955094, 0.6375291637263358, -0.0037645818631678873
    };
    const int n = (int)(sizeof knots / sizeof knots[0]);
    const double u = x;
    /* Interval i holds u (the end intervals continue outside the knots) */
    int i = 0;
    while (i < n - 2 && u >= knots[i + 1]) i++;
    const double h = knots[i + 1] - knots[i];
    const double s = (u - knots[i]) / h;
    const double s2 = s * s;
    const double s3 = s2 * s;
    const double v = (2.0 * s3 - 3.0 * s2 + 1.0) * values[i] + (s3 - 2.0 * s2 + s) * h * slopes[i]
        + (-2.0 * s3 + 3.0 * s2) * values[i + 1] + (s3 - s2) * h * slopes[i + 1];
    return v;
}

==== piecewise ====
/*
 * Curve Digitizer fit: Piecewise linear, 3 segments
 * Dataset: Derating
 * Valid for -20 <= x <= 125, the range of the 50 fitted points; outside it the curve is extrapolated
 * R^2 = 0.99993, RMSE = 0.19, AICc = -151.1
 */
#include <math.h>

double derating(double x)
{
    static const double knots[] = {-20.0, 40.0, 85.0, 125.0};
    static const double values[] = {100.0, 100.2, 60.0, 0.0};
    const int n = (int)(sizeof knots / sizeof knots[0]);
    const double u = x;
    /* Interval i holds u (the end intervals continue outside the knots) */
    int i = 0;
    while (i < n - 2 && u >= knots[i + 1]) i++;
    const double s = (u - knots[i]) / (knots[i + 1] - knots[i]);
    const double v = values[i] + s * (values[i + 1] - values[i]);
    return v;
}
