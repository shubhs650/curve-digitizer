==== spline ====
// Curve Digitizer fit: Cubic spline, 7 knots
// Dataset: Torque
// Valid for 0 <= x <= 10, the range of the 80 fitted points; outside it the curve is extrapolated
// R^2 = 0.99841, RMSE = 0.0287, AICc = -540.3
function torque(x) {
  const u = x;
  const knots = [0, 1.25, 2.5, 4, 5.5, 8, 10];
  const values = [1, 2.1, 1.6, 0.2, -1.05, 0.5, 0.92];
  const slopes = [
    1.189899854508072, 0.26020029098385655, -0.7907010184434978, -1.073155868029238,
    -0.21667550943955094, 0.6375291637263358, -0.0037645818631678873
  ];
  // Interval i holds u (the end intervals continue outside the knots)
  let i = 0;
  while (i < knots.length - 2 && u >= knots[i + 1]) i++;
  const h = knots[i + 1] - knots[i];
  const s = (u - knots[i]) / h;
  const s2 = s * s;
  const s3 = s2 * s;
  const v =
    (2 * s3 - 3 * s2 + 1) * values[i] + (s3 - 2 * s2 + s) * h * slopes[i] +
    (-2 * s3 + 3 * s2) * values[i + 1] + (s3 - s2) * h * slopes[i + 1];
  return v;
}

==== piecewise ====
// Curve Digitizer fit: Piecewise linear, 3 segments
// Dataset: Derating
// Valid for -20 <= x <= 125, the range of the 50 fitted points; outside it the curve is extrapolated
// R^2 = 0.99993, RMSE = 0.19, AICc = -151.1
function derating(x) {
  const u = x;
  const knots = [-20, 40, 85, 125];
  const values = [100, 100.2, 60, 0];
  // Interval i holds u (the end intervals continue outside the knots)
  let i = 0;
  while (i < knots.length - 2 && u >= knots[i + 1]) i++;
  const s = (u - knots[i]) / (knots[i + 1] - knots[i]);
  const v = values[i] + s * (values[i + 1] - values[i]);
  return v;
}

==== piecewise-logx ====
// Curve Digitizer fit: Piecewise linear, 2 segments
// Dataset: Fatigue
// Fitted on log paper: u = log10(x), y = v
// Valid for 1000 <= x <= 100000000, the range of the 33 fitted points; outside it the curve is extrapolated
// R^2 = 0.9958, RMSE = 4.1, AICc = 101.7
function fatigue(x) {
  const u = x > 0 ? Math.log10(x) : NaN;
  const knots = [3, 6, 8];
  const values = [450, 210, 205];
  // Interval i holds u (the end intervals continue outside the knots)
  let i = 0;
  while (i < knots.length - 2 && u >= knots[i + 1]) i++;
  const s = (u - knots[i]) / (knots[i + 1] - knots[i]);
  const v = values[i] + s * (values[i + 1] - values[i]);
  return v;
}
