==== spline ====
function y = torque(x)
%TORQUE Curve Digitizer fit: Cubic spline, 7 knots
%   Dataset: Torque
%   Valid for 0 <= x <= 10, the range of the 80 fitted points; outside it the curve is extrapolated
%   R^2 = 0.99841, RMSE = 0.0287, AICc = -540.3
%   x may be an array of any shape; y has the same shape (NaN where the fit is undefined).
    sz = size(x);
    x = x(:);
    u = x;
    knots = [0, 1.25, 2.5, 4, 5.5, 8, 10].';
    values = [1, 2.1, 1.6, 0.2, -1.05, 0.5, 0.92].';
    slopes = [ ...
        1.189899854508072, 0.26020029098385655, -0.7907010184434978, -1.073155868029238, ...
        -0.21667550943955094, 0.6375291637263358, -0.0037645818631678873 ...
    ].';
    % Interval i holds u (the end intervals continue outside the knots)
    i = sum(u >= knots(2:end-1).', 2) + 1;
    h = knots(i + 1) - knots(i);
    s = (u - knots(i)) ./ h;
    s2 = s .* s;
    s3 = s2 .* s;
    v = (2 .* s3 - 3 .* s2 + 1) .* values(i) + (s3 - 2 .* s2 + s) .* h .* slopes(i) ...
        + (-2 .* s3 + 3 .* s2) .* values(i + 1) + (s3 - s2) .* h .* slopes(i + 1);
    y = reshape(v, sz);
end

function y = derating(x)
%DERATING Curve Digitizer fit: Piecewise linear, 3 segments
%   Dataset: Derating
%   Valid for -20 <= x <= 125, the range of the 50 fitted points; outside it the curve is extrapolated
%   R^2 = 0.99993, RMSE = 0.19, AICc = -151.1
%   x may be an array of any shape; y has the same shape (NaN where the fit is undefined).
    sz = size(x);
    x = x(:);
    u = x;
    knots = [-20, 40, 85, 125].';
    values = [100, 100.2, 60, 0].';
    % Interval i holds u (the end intervals continue outside the knots)
    i = sum(u >= knots(2:end-1).', 2) + 1;
    s = (u - knots(i)) ./ (knots(i + 1) - knots(i));
    v = values(i) + s .* (values(i + 1) - values(i));
    y = reshape(v, sz);
end
