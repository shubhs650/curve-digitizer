==== spline ====
' Curve Digitizer fit: Cubic spline, 7 knots
' Dataset: Torque
' Valid for 0 <= x <= 10, the range of the 80 fitted points; outside it the curve is extrapolated
' R^2 = 0.99841, RMSE = 0.0287, AICc = -540.3
Public Function torque(ByVal x As Double) As Variant
    Dim u As Double, v As Double
    Dim knots As Variant, values As Variant, slopes As Variant
    Dim i As Long, h As Double, s As Double, s2 As Double, s3 As Double
    u = x
    knots = Array(0#, 1.25, 2.5, 4#, 5.5, 8#, 10#)
    values = Array(1#, 2.1, 1.6, 0.2, -1.05, 0.5, 0.92)
    slopes = Array( _
        1.189899854508072, 0.26020029098385655, -0.7907010184434978, -1.073155868029238, _
        -0.21667550943955094, 0.6375291637263358, -0.0037645818631678873)
    ' Interval i holds u (the end intervals continue outside the knots)
    i = LBound(knots)
    Do While i < UBound(knots) - 1
        If u < knots(i + 1) Then Exit Do
        i = i + 1
    Loop
    h = knots(i + 1) - knots(i)
    s = (u - knots(i)) / h
    s2 = s * s
    s3 = s2 * s
    v = (2 * s3 - 3 * s2 + 1) * values(i) + (s3 - 2 * s2 + s) * h * slopes(i) _
        + (-2 * s3 + 3 * s2) * values(i + 1) + (s3 - s2) * h * slopes(i + 1)
    torque = v
End Function

==== piecewise ====
' Curve Digitizer fit: Piecewise linear, 3 segments
' Dataset: Derating
' Valid for -20 <= x <= 125, the range of the 50 fitted points; outside it the curve is extrapolated
' R^2 = 0.99993, RMSE = 0.19, AICc = -151.1
Public Function derating(ByVal x As Double) As Variant
    Dim u As Double, v As Double
    Dim knots As Variant, values As Variant
    Dim i As Long, h As Double, s As Double
    u = x
    knots = Array(-20#, 40#, 85#, 125#)
    values = Array(100#, 100.2, 60#, 0#)
    ' Interval i holds u (the end intervals continue outside the knots)
    i = LBound(knots)
    Do While i < UBound(knots) - 1
        If u < knots(i + 1) Then Exit Do
        i = i + 1
    Loop
    s = (u - knots(i)) / (knots(i + 1) - knots(i))
    v = values(i) + s * (values(i + 1) - values(i))
    derating = v
End Function
